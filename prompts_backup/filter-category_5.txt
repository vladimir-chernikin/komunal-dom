### Шаг6. Определение победителя
Z1_cat = категория с максимальным значением в M_CANDIDATE
Z1 = M_CANDIDATE[Z1_cat]
total = СУММА всех значений из M_CANDIDATE
ostatok = total - Z1
category = null
category_confidence = "0.5"

# Условие: Z1 > остаток * 0.7 (значимый отрыв)
ЕСЛИ Z1 > ostatok * 0.7 И Z1 > 0.3:
    category = Z1_cat
    category_confidence = СТРОКА(ОКРУГЛИТЬ(MIN(Z1, 1.0), 1))
    reasoning_txt = reasoning_txt + " | Шаг6: Z1=" + Z1_cat + "(" + СТРОКА(Z1) + "), ostatok=" + СТРОКА(ОКРУГЛИТЬ(ostatok, 3)) + ", Z1>ostatok*0.7=" + СТРОКА(ОКРУГЛИТЬ(ostatok*0.7, 3)) + " → " + category + "(" + category_confidence + ")"
ИНАЧЕ:
    reasoning_txt = reasoning_txt + " | Шаг6: Z1=" + Z1_cat + "(" + СТРОКА(Z1) + ") недостаточно → null(0.5)"
